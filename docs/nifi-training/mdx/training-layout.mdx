    <script>
      const documents = JSON.parse(document.getElementById("documents-data").textContent);
      const toc = document.getElementById("toc");
      const article = document.getElementById("article");
      const search = document.getElementById("search");
      const currentFile = document.getElementById("current-file");
      const previous = document.getElementById("previous");
      const next = document.getElementById("next");

      let currentIndex = 0;
      let visibleDocuments = documents;

      function setDocument(index, shouldPushHash = true) {
        const documentItem = visibleDocuments[index] || documents[0];
        currentIndex = documents.findIndex((item) => item.id === documentItem.id);
        article.innerHTML = documentItem.html;
        currentFile.textContent = documentItem.file;

        for (const button of toc.querySelectorAll("button")) {
          button.setAttribute("aria-current", button.dataset.id === documentItem.id ? "page" : "false");
        }

        if (shouldPushHash) {
          history.replaceState(null, "", `#${documentItem.id}`);
        }

        previous.disabled = currentIndex <= 0;
        next.disabled = currentIndex >= documents.length - 1;
        article.scrollTop = 0;
        window.scrollTo({ top: 0, behavior: "auto" });
      }

      function renderToc() {
        toc.innerHTML = "";

        if (visibleDocuments.length === 0) {
          toc.innerHTML = '<div class="empty">沒有符合的文件</div>';
          return;
        }

        for (const documentItem of visibleDocuments) {
          const button = document.createElement("button");
          button.type = "button";
          button.className = "toc__button";
          button.dataset.id = documentItem.id;
          button.appendChild(document.createTextNode(documentItem.title));

          const file = document.createElement("span");
          file.className = "toc__file";
          file.textContent = documentItem.file;
          button.appendChild(file);

          button.addEventListener("click", () => {
            const nextIndex = visibleDocuments.findIndex((item) => item.id === documentItem.id);
            setDocument(nextIndex);
          });
          toc.appendChild(button);
        }
      }

      search.addEventListener("input", () => {
        const keyword = search.value.trim().toLowerCase();
        visibleDocuments = documents.filter((item) => {
          return item.title.toLowerCase().includes(keyword) || item.file.toLowerCase().includes(keyword);
        });
        renderToc();
        const activeInResult = visibleDocuments.findIndex((item) => item.id === documents[currentIndex].id);
        if (activeInResult >= 0) {
          setDocument(activeInResult, false);
        }
      });

      previous.addEventListener("click", () => {
        if (currentIndex > 0) {
          visibleDocuments = documents;
          search.value = "";
          renderToc();
          setDocument(currentIndex - 1);
        }
      });

      next.addEventListener("click", () => {
        if (currentIndex < documents.length - 1) {
          visibleDocuments = documents;
          search.value = "";
          renderToc();
          setDocument(currentIndex + 1);
        }
      });

      window.addEventListener("hashchange", () => {
        const nextId = decodeURIComponent(location.hash.replace(/^#/, ""));
        const nextIndex = documents.findIndex((item) => item.id === nextId);
        if (nextIndex >= 0) {
          visibleDocuments = documents;
          search.value = "";
          renderToc();
          setDocument(nextIndex, false);
        }
      });

      renderToc();

      const initialId = decodeURIComponent(location.hash.replace(/^#/, ""));
      const initialIndex = documents.findIndex((item) => item.id === initialId);
      setDocument(initialIndex >= 0 ? initialIndex : 0, initialIndex >= 0);
    </script>
